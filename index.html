#region Using declarations using System; using System.ComponentModel; using System.Windows.Media; using NinjaTrader.NinjaScript; using NinjaTrader.Data; using NinjaTrader.Cbi; using NinjaTrader.NinjaScript.Indicators; using NinjaTrader.NinjaScript.DrawingTools; #endregion namespace NinjaTrader.NinjaScript.Indicators { public class BCL_NQ_Radar_V2 : Indicator { private string htfBias = "Neutral"; private bool sweepDetected = false; private bool ftdConfirmed = false; private bool ifvgDetected = false; private double ifvgMid = 0; private DateTime lastAlertTime = DateTime.MinValue; protected override void OnStateChange() { if (State == State.SetDefaults) { Name = "BCL_NQ_Radar_V2"; Calculate = Calculate.OnBarClose; IsOverlay = true; ProximityTicks = 10; // Ticks de distancia para activar el radar/sonido SoundFile = "Alert1.wav"; } else if (State == State.Configure) { AddDataSeries(BarsPeriodType.Minute, 60); // 1H AddDataSeries(BarsPeriodType.Minute, 240); // 4H } } protected override void OnBarUpdate() { if (CurrentBars[0] < 20 || CurrentBars[1] < 20 || CurrentBars[2] < 20) return; if (Bars.IsFirstBarOfSession) { sweepDetected = false; ftdConfirmed = false; ifvgDetected = false; ifvgMid = 0; } // Evaluar únicamente en la serie primaria de ejecución (3 Minutos) if (BarsInProgress != 0) return; // Filtro de horario (9:30 AM - 11:00 AM) if (ToTime(Time[0]) < 93000 || ToTime(Time[0]) > 110000) return; // HTF Bias double close1H = Closes[1][0]; double close4H = Closes[2][0]; if (close1H < SMA(Closes[1], 20)[0] && close4H < SMA(Closes[2], 20)[0]) htfBias = "Bearish"; else if (close1H > SMA(Closes[1], 20)[0] && close4H > SMA(Closes[2], 20)[0]) htfBias = "Bullish"; else htfBias = "Neutral"; if (htfBias == "Neutral") return; // Liquidez double equalHigh = MAX(High, 10)[1]; double equalLow = MIN(Low, 10)[1]; if (htfBias == "Bearish" && High[0] > equalHigh && Close[0] < equalHigh) sweepDetected = true; if (htfBias == "Bullish" && Low[0] < equalLow && Close[0] > equalLow) sweepDetected = true; if (!sweepDetected) return; // FTD if (htfBias == "Bearish" && Close[0] < Open[0]) ftdConfirmed = true; if (htfBias == "Bullish" && Close[0] > Open[0]) ftdConfirmed = true; if (!ftdConfirmed) return; // Displacement + IFVG double body = Math.Abs(Close[0] - Open[0]); double avgBody = SMA(Math.Abs(Close[0] - Open[0]), 10)[0]; if (body > avgBody * 1.5) { if (htfBias == "Bearish" && Low[0] > High[2]) { ifvgDetected = true; ifvgMid = (Low[0] + High[2]) / 2; } if (htfBias == "Bullish" && High[0] < Low[2]) { ifvgDetected = true; ifvgMid = (High[0] + Low[2]) / 2; } } // RADAR & ALERTA SONORA if (ifvgDetected && ifvgMid > 0) { double distanceInTicks = Math.Abs(Close[0] - ifvgMid) / TickSize; // Dibujar línea del radar en el gráfico Draw.Line(this, "IFVG_Line_" + CurrentBar, false, 2, ifvgMid, 0, ifvgMid, htfBias == "Bullish" ? Brushes.LimeGreen : Brushes.Red, DashStyleHelper.Solid, 2); // Activar sonido si el precio está dentro de la distancia configurada if (distanceInTicks <= ProximityTicks && Time[0] != lastAlertTime) { PlaySound(NinjaTrader.Core.Globals.InstallDir + @"\sounds\" + SoundFile); Draw.Text(this, "RadarAlert_" + CurrentBar, "ZONA DE ENTRADA (" + htfBias + ")", 0, High[0] + (10 * TickSize), htfBias == "Bullish" ? Brushes.LimeGreen : Brushes.Red); lastAlertTime = Time[0]; } } } #region Properties [NinjaScriptProperty] [Range(1, 100)] [Display(Name = "Proximidad en Ticks", Description = "Distancia en ticks hacia ifvgMid para activar la alerta", GroupName = "Radar Config", Order = 1)] public int ProximityTicks { get; set; } [NinjaScriptProperty] [Display(Name = "Archivo de Sonido", Description = "Nombre del archivo .wav en la carpeta /sounds/", GroupName = "Radar Config", Order = 2)] public string SoundFile { get; set; } #endregion } }
